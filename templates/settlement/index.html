completed_months = min(months(agreement → today), tenure)

future_principal = future_due_count × (loan_amount ÷ tenure)
overdue_count = max(0, completed_months − paid_months)
total_overdue = overdue_count × emi
fc_charges = future_principal × 5%
settlement = future_principal + total_overdue + fc_charges
